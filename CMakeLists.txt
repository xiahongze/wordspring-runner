cmake_minimum_required(VERSION 3.24)
project(wordspring-runner VERSION 0.2.0 LANGUAGES CXX)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
option(WORDSPRING_BUILD_KDE "Build the Plasma KRunner plugin and Qt Widgets application" ON)
option(WORDSPRING_INSTALL_NOCTALIA "Install the Noctalia Luau frontend" ON)

find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(SQLite3 REQUIRED)
# CMake before 4.1 names the imported target SQLite::SQLite3.
if(NOT TARGET SQLite3::SQLite3)
  add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()

add_library(wordspring_core STATIC src/store.cpp src/store.h src/userstate.cpp src/userstate.h
  src/core/configuration.cpp src/core/configuration.h src/core/articleformat.cpp src/core/articleformat.h)
set_target_properties(wordspring_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(wordspring_core PUBLIC Qt6::Core SQLite3::SQLite3)
target_include_directories(wordspring_core PUBLIC src)

add_executable(wordspring-lookup src/cli/main.cpp)
target_link_libraries(wordspring-lookup PRIVATE wordspring_core)
target_compile_definitions(wordspring-lookup PRIVATE WORDSPRING_VERSION="${PROJECT_VERSION}")
target_compile_options(wordspring-lookup PRIVATE -fPIC)
install(TARGETS wordspring-lookup RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(WORDSPRING_BUILD_KDE)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(KF6Runner REQUIRED)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6I18n REQUIRED)
set(KDE_INSTALL_PLUGINDIR ${CMAKE_INSTALL_LIBDIR}/qt6/plugins)

kcoreaddons_add_plugin(krunner_wordspring SOURCES src/runner.cpp src/runner.h INSTALL_NAMESPACE "kf6/krunner")
set_target_properties(krunner_wordspring PROPERTIES PREFIX "")
target_link_libraries(krunner_wordspring PRIVATE wordspring_core KF6::Runner KF6::I18n)
target_compile_definitions(krunner_wordspring PRIVATE WORDSPRING_APP_PATH="${CMAKE_INSTALL_FULL_BINDIR}/wordspring")

add_library(wordspring_settings STATIC src/settingswidget.cpp src/settingswidget.h)
set_target_properties(wordspring_settings PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(wordspring_settings PUBLIC wordspring_core Qt6::Widgets KF6::I18n)
target_compile_definitions(wordspring_settings PRIVATE WORDSPRING_SOURCE_IMPORTER="${CMAKE_SOURCE_DIR}/importer/import_dictionary.py")

add_library(wordspring_view STATIC src/articlewindow.cpp src/articlewindow.h)
set_target_properties(wordspring_view PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(wordspring_view PUBLIC wordspring_core Qt6::Widgets KF6::I18n)

kcoreaddons_add_plugin(kcm_krunner_wordspring SOURCES src/kcm.cpp INSTALL_NAMESPACE "kf6/krunner/kcms")
set_target_properties(kcm_krunner_wordspring PROPERTIES PREFIX "")
target_link_libraries(kcm_krunner_wordspring PRIVATE wordspring_settings KF6::KCMUtils)

add_executable(wordspring src/main.cpp)
target_link_libraries(wordspring PRIVATE wordspring_settings wordspring_view Qt6::Widgets KF6::I18n)
# PIC (not PIE) avoids copy relocations, so a binary built against one distro's Qt also runs where Qt
# is built with -mno-direct-extern-access, as on Arch.
target_compile_options(wordspring PRIVATE -fPIC)
install(TARGETS wordspring RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(FILES importer/import_dictionary.py DESTINATION ${CMAKE_INSTALL_DATADIR}/wordspring)
if(WORDSPRING_INSTALL_NOCTALIA)
  install(DIRECTORY wordspring/ DESTINATION ${CMAKE_INSTALL_DATADIR}/noctalia/plugins/wordspring)
endif()

if(BUILD_TESTING)
  add_executable(store_test tests/store_test.cpp)
  target_link_libraries(store_test PRIVATE wordspring_core Qt6::Test)
  add_test(NAME store_test COMMAND store_test)
  if(WORDSPRING_BUILD_KDE)
  add_executable(configuration_test tests/configuration_test.cpp)
  target_link_libraries(configuration_test PRIVATE wordspring_core Qt6::Test KF6::ConfigCore)
  add_test(NAME configuration_test COMMAND configuration_test)
  add_executable(articlewindow_test tests/articlewindow_test.cpp)
  target_link_libraries(articlewindow_test PRIVATE wordspring_view Qt6::Test)
  add_test(NAME articlewindow_test COMMAND articlewindow_test)
  set_tests_properties(articlewindow_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic;QT_STYLE_OVERRIDE=Fusion")
  add_executable(plugin_test tests/plugin_test.cpp)
  target_link_libraries(plugin_test PRIVATE Qt6::Core)
  target_compile_definitions(plugin_test PRIVATE WORDSPRING_RUNNER_PLUGIN="$<TARGET_FILE:krunner_wordspring>" WORDSPRING_KCM_PLUGIN="$<TARGET_FILE:kcm_krunner_wordspring>")
  add_dependencies(plugin_test krunner_wordspring kcm_krunner_wordspring)
  add_test(NAME plugin_test COMMAND plugin_test)
  endif()
  add_test(NAME cli_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:wordspring-lookup>)
  add_test(NAME noctalia_manifest_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_noctalia.py)
  find_program(WORDSPRING_LUAU_EXECUTABLE NAMES luau)
  if(WORDSPRING_LUAU_EXECUTABLE)
    add_test(NAME noctalia_runtime_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_noctalia.py ${WORDSPRING_LUAU_EXECUTABLE})
    set_tests_properties(noctalia_runtime_test PROPERTIES TIMEOUT 30)
  endif()
  find_program(WORDSPRING_NOCTALIA_EXECUTABLE NAMES noctalia)
  if(WORDSPRING_NOCTALIA_EXECUTABLE)
    add_test(NAME noctalia_lint COMMAND ${WORDSPRING_NOCTALIA_EXECUTABLE} plugins lint ${CMAKE_SOURCE_DIR}/wordspring)
  endif()
  add_test(NAME importer_test COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_importer.py -v)
  set_tests_properties(importer_test PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  set_tests_properties(store_test cli_test importer_test noctalia_manifest_test PROPERTIES TIMEOUT 30)
  if(WORDSPRING_BUILD_KDE)
    set_tests_properties(configuration_test articlewindow_test plugin_test PROPERTIES TIMEOUT 30)
  endif()
endif()
